-- Batch ETL procedure with exception handling, reconciliation & audit logging CREATE OR REPLACE PROCEDURE process_daily_batch ( p_load_date IN DATE, p_batch_id IN NUMBER, p_rows_loaded OUT NUMBER ) AS -- Bulk collect for high-volume processing TYPE t_staging IS TABLE OF staging_table%ROWTYPE; l_records t_staging; l_errors NUMBER := 0; l_inserted NUMBER := 0; CURSOR c_stage IS SELECT s.* FROM staging_table s WHERE s.load_date = p_load_date AND s.batch_id = p_batch_id AND s.status = 'PENDING' ORDER BY s.record_seq; BEGIN OPEN c_stage; LOOP FETCH c_stage BULK COLLECT INTO l_records LIMIT 500; EXIT WHEN l_records.COUNT = 0; FORALL i IN 1..l_records.COUNT SAVE EXCEPTIONS INSERT INTO target_table ( record_id, source_key, transformed_val, load_date, batch_id, created_at ) VALUES ( l_records(i).id, l_records(i).source_key, TRIM(UPPER(l_records(i).raw_value)), p_load_date, p_batch_id, SYSDATE ); l_inserted := l_inserted + SQL%ROWCOUNT; END LOOP; CLOSE c_stage; p_rows_loaded := l_inserted; COMMIT; EXCEPTION WHEN OTHERS THEN ROLLBACK; log_error('process_daily_batch', SQLERRM, p_batch_id); RAISE; END process_daily_batch;
